        // Shuffles an array randomly.

        function shuffleArray(array) {

            for (let i = array.length - 1; i > 0; i--) {

                const j = Math.floor(Math.random() * (i + 1));

                [array[i], array[j]] = [array[j], array[i]];

            }

            return array;

        }

        }


        // Renders the god options for the first phase of the round.

        function renderGods() {

            choicesContainer.innerHTML = '';

            shuffleArray(allGods).forEach(god => {

                godElement.classList.add('choice-container', 'cursor-pointer');

                godElement.innerHTML = `

                    <p class="choice-label">${god.replace(/_/g, ' ')}</p>

                godElement.addEventListener('click', () => handleGodClick(god));

                choicesContainer.appendChild(godElement);

            });

        }

        // Handles the player's god selection.

        function handleGodClick(selectedGod) {

            const currentClue = shuffledGameData[currentClueIndex];

            if (currentPhase !== 'god_selection') return;


            if (currentClue.god === selectedGod) {

                showMessage('Correct! Now, make your sacrifice.', 'text-green-400');

                score++;

                scoreEl.textContent = score;

                currentPhase = 'sacrifice_selection';

                

                // Show the correct god's name and image

                godNameEl.textContent = currentClue.god;

                godImageEl.src = godImages[currentClue.god];

                godImageEl.alt = currentClue.god;

                godImageEl.classList.remove('hidden');

                

                // Transition the UI to the sacrifice phase

                renderSacrifices();

                altarArea.classList.remove('hidden');

                godDisplayArea.classList.remove('hidden');


            } else {

                score--;

                scoreEl.textContent = score;

                showIncorrectFeedback('That is not the correct deity for this situation. Consult your priests to discern the will of the gods.');

            }

        }


        // Handles the player's sacrifice drop.

        function handleDrop(e) {

            e.preventDefault();

            if (currentPhase !== 'sacrifice_selection') return;

            const droppedSacrifice = e.dataTransfer.getData('text/plain');

            const currentClue = shuffledGameData[currentClueIndex];


            if (currentClue.sacrifice === droppedSacrifice) {

                showMessage('The gods are pleased! Your devotion is absolute.', 'text-green-400');

                score++;

                scoreEl.textContent = score;

                

                altarEl.innerHTML = sacrificeSVGs[droppedSacrifice];

                altarEl.classList.add('border-green-400');

                

                setTimeout(() => {

                    altarEl.classList.remove('border-green-400');

                    endRound();

                }, 2000);

            } else {

                score--;

                scoreEl.textContent = score;

                altarEl.innerHTML = `<p class="text-lg md:text-xl text-gray-400 select-none">Drag Sacrifice Here</p>`;

                showIncorrectFeedback('The gods are displeased with your offering. It is not the sacrifice they require for this ritual.');

            }

        }

        

        // Advances to the next round or ends the game.

        function endRound() {

            currentClueIndex++;

            currentPhase = 'god_selection';

            

            setTimeout(() => {

                // End the game after 4 rounds

                if (currentClueIndex >= MAX_ROUNDS) {

                    endGame();

                } else {

                    nextRound();

                }

            }, 1000);

        }


        // Sets up a new round.

        function nextRound() {

            const currentClue = shuffledGameData[currentClueIndex];

            clueEl.textContent = currentClue.clue;

            roundEl.textContent = `${currentClueIndex + 1}/${MAX_ROUNDS}`;

            

            // Reset UI for new round

            godNameEl.textContent = '';

            godImageEl.classList.add('hidden');

            altarArea.classList.add('hidden');

            godDisplayArea.classList.remove('hidden');

            altarEl.innerHTML = `<p class="text-lg md:text-xl text-gray-400 select-none">Drag Sacrifice Here</p>`;

            

            // Start with god selection

            renderGods();

        }


        // Starts a new game.

        function startGame() {

            shuffledGameData = shuffleArray([...gameData]);

            currentClueIndex = 0;

            score = 0;

            scoreEl.textContent = score;

            startScreen.classList.add('hidden');

            gameUI.classList.remove('hidden');

            gameOverScreen.classList.add('hidden');

            incorrectFeedbackScreen.classList.add('hidden');

            nextRound();

        }

        

        // Ends the game and displays the final score.

        function endGame() {

            gameUI.classList.add('hidden');

            gameOverScreen.classList.remove('hidden');

            finalScoreEl.textContent = score;

        }


        // Returns the player to the previous state after an incorrect choice.

        function goBack() {

            incorrectFeedbackScreen.classList.add('hidden');

            gameUI.classList.remove('hidden');


            if (currentPhase === 'god_selection') {

                renderGods();

            } else if (currentPhase === 'sacrifice_selection') {

                renderSacrifices();

                godDisplayArea.classList.remove('hidden');

                altarArea.classList.remove('hidden');

            }

        }

        

        // Event Listeners

        startButton.addEventListener('click', startGame);

        restartButton.addEventListener('click', startGame);

        backButton.addEventListener('click', goBack);

        newQuestButton.addEventListener('click', startGame);

        altarEl.addEventListener('dragover', e => e.preventDefault());

        altarEl.addEventListener('drop', handleDrop);

        document.addEventListener('dragstart', e => {

            const target = e.target.closest('.choice-container');

            if (target && currentPhase === 'sacrifice_selection') {

                e.dataTransfer.setData('text/plain', target.dataset.sacrifice);

            }

        });
